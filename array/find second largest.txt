
const max=(arr)=>{
    let largest = -Infinity;
    let secondLargest = -Infinity;

    arr.forEach((curr,index)=>{
        if(largest<curr){
            secondLargest = largest;
            largest = curr;
        }
        else if(largest>curr && secondLargest < curr){
            secondLargest = curr; 
        }

        console.log(`${index}=>`,largest, secondLargest)

    })
  return secondLargest
}
