function reverse(arr) {
    let left = 0;
    let right = arr.length - 1;

    while (left < right) {
        //--------
        [arr[left], arr[right]] = [arr[right], arr[left]];
        /// or 
        [letftValue, rightValue] = [arr[right], arr[left]];
        arr[left] = letftValue;
        arr[right] = rightValue;
        //-------------
        left++;
        right--;
    }

    return arr;
}

const arr = [1, 2, 3, 4, 6];

console.log(reverse(arr));
