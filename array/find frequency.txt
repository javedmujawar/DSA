//Find the sum of array elements
function findFrequency(arr) {
    return arr.reduce((acc, cur) => {
        if(!acc[cur]) acc[cur] = 0;
         acc[cur]= acc[cur]+1;
         return acc;
    }
        , {});
}
const arr = [1, 1, 2, 2, 3, 4, 6];
console.log(findFrequency(arr));
